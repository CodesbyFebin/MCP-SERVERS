## Implementation

import { describe, it, expect } from 'vitest'

describe('MCP Server', () => {
  it('should list tools', async () => {
    const server = new Server({ name: 'test', version: '1.0.0' }, { capabilities: { tools: {} } })
    const tools = await server.listTools()

  })
})
